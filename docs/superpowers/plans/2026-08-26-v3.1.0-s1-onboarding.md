> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现桌面应用首次启动 4 步向导：欢迎 → 创建管理员 → 下载目录 → 完成。First-run 双重判定（哨兵文件 + users 表为空），支持跳过向导。

**Tech Stack:** Electron 32、React 19、Zod、Express、bcrypt、Prisma 7、Vite 8、Vitest。

```typescript
import { Request, Response } from 'express';
import bcrypt from 'bcryptjs';
import jwt from 'jsonwebtoken';
import { prisma } from '../lib/prisma.js';
import { isUsersTableEmpty } from '../lib/check-first-run.js';
import { OnboardingStep2Schema } from '@dm/shared';

const JWT_SECRET = process.env.JWT_SECRET || 'dev-secret';

export async function setup(req: Request, res: Response) {
  // Gate: only when users table is empty
  const empty = await isUsersTableEmpty();
  if (!empty) {
    return res.status(409).json({
      success: false,
      error: { code: 'ALREADY_ONBOARDED', message: 'Setup is only available on first run' },
    });
  }

  // Validate input
  const parsed = OnboardingStep2Schema.safeParse(req.body);
  if (!parsed.success) {
    return res.status(400).json({
      success: false,
      error: { code: 'VALIDATION_ERROR', message: parsed.error.issues[0]?.message },
    });
  }

  const { username, email, password } = parsed.data;
  const hashed = await bcrypt.hash(password, 10);

  const user = await prisma.user.create({
    data: {
      username,
      email,
      password: hashed,
      role: 'ADMIN',
    },
    select: { id: true, username: true, email: true, role: true },
  });

  const token = jwt.sign({ sub: user.id, role: user.role }, JWT_SECRET, { expiresIn: '7d' });

  return res.json({
    success: true,
    data: {
      token,
      user,
    },
  });
}
```

```typescript
import path from 'node:path';
import fs from 'fs-extra';
import { app } from 'electron';

export interface FirstRunDeps {
  countUsers: () => Promise<number>;
}

export function getSentinelPath(): string {
  return path.join(app.getPath('userData'), '.onboarded');
}

/**
 * Returns true on first run.
 * Two-factor check: sentinel file AND users table empty.
 * Either signal alone is enough to consider "already onboarded":
 *   - Sentinel exists → user explicitly completed onboarding
 *   - Users exist → DB seeded externally (treat as onboarded)
 */
export async function isFirstRun(deps: FirstRunDeps): Promise<boolean> {
  if (await fs.pathExists(getSentinelPath())) return false;
  const userCount = await deps.countUsers().catch(() => 0);
  return userCount === 0;
}

export async function markOnboarded(): Promise<void> {
  await fs.writeFile(getSentinelPath(), new Date().toISOString());
}

export async function skipOnboarding(): Promise<void> {
  // Equivalent to markOnboarded but explicitly tagged as "skipped"
  await fs.writeFile(getSentinelPath(), `skipped:${new Date().toISOString()}`);
}
```

```typescript
export type OnboardingFlow = 'running' | 'complete' | 'skipped';

export interface OnboardingState {
  step: 1 | 2 | 3 | 4;
  flow: OnboardingFlow;
}

export interface OnboardingMachine {
  state: OnboardingState;
  next: () => void;
  back: () => void;
  skip: () => void;
  reset: () => void;
}

export function createOnboardingMachine(initial: OnboardingState = { step: 1, flow: 'running' }): OnboardingMachine {
  const state: OnboardingState = { ...initial };

  return {
    get state() {
      return { ...state };
    },
    next() {
      if (state.flow !== 'running') return;
      if (state.step < 4) {
        state.step = (state.step + 1) as OnboardingState['step'];
      } else {
        state.flow = 'complete';
      }
    },
    back() {
      if (state.flow !== 'running') return;
      if (state.step > 1) {
        state.step = (state.step - 1) as OnboardingState['step'];
      }
    },
    skip() {
      state.flow = 'skipped';
    },
    reset() {
      state.step = 1;
      state.flow = 'running';
    },
  };
}
```

```typescript
  user: { id: number; username: string; email: string; role: string };
}

export async function callSetup(input: OnboardingStep2Input): Promise<SetupResponse> {
  const res = await fetch(`${API_BASE}/api/auth/setup`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(input),
  });
  const json = await res.json();
  if (!json.success) {
    throw new Error(json.error?.message || 'Setup failed');
  }
  return json.data;
}

export async function isFirstRun(): Promise<boolean> {
  // Use Electron bridge if available; otherwise check via API
  if (typeof (window as any).onboarding?.isFirstRun === 'function') {
    return (window as any).onboarding.isFirstRun();
  }
  // Fallback: check users count via public API
  const res = await fetch(`${API_BASE}/api/auth/users/count`);
  const json = await res.json();
  return json.data?.count === 0;
}

export async function markOnboarded(): Promise<void> {
  if (typeof (window as any).onboarding?.markDone === 'function') {
    await (window as any).onboarding.markDone();
  }
}

export async function skipOnboarding(): Promise<void> {
  if (typeof (window as any).onboarding?.skip === 'function') {
    await (window as any).onboarding.skip();
  }
}
```

```tsx
        setChecking(false);
      });

    }

      <Suspense fallback={<div>Loading...</div>}>
        <Routes>
          {showOnboarding && <Route path="/*" element={<Navigate to="/onboarding" replace />} />}
          <Route path="/onboarding" element={<OnboardingPage />} />
          <Route path="/downloads" element={<DownloadsPage />} />
          <Route path="*" element={<Navigate to="/downloads" replace />} />
        </Routes>
      </Suspense>
    </BrowserRouter>
  );
}
```

```typescript
import { Response, Request } from 'express';
import bcrypt from 'bcryptjs';
import jwt from 'jsonwebtoken';
import { prisma } from '../lib/prisma.js';
import { isUsersTableEmpty } from '../lib/check-first-run.js';

const JWT_SECRET = process.env.JWT_SECRET || 'dev-secret';
const DEFAULT_ADMIN_USERNAME = process.env.DEFAULT_ADMIN_USERNAME || 'admin';
const DEFAULT_ADMIN_PASSWORD = process.env.DEFAULT_ADMIN_PASSWORD || 'Admin@123';
const DEFAULT_ADMIN_EMAIL = process.env.DEFAULT_ADMIN_EMAIL || 'admin@dm.local';

/**
 * POST /auth/setup/skip
 * Skip onboarding wizard with default admin credentials.
 * Frontend will show a warning banner: "Default password in use - change it".
 */
export async function setupSkip(_req: Request, res: Response) {
  const empty = await isUsersTableEmpty();
  if (!empty) {
    return res.status(409).json({
      success: false,
      error: { code: 'ALREADY_ONBOARDED', message: 'Setup is only available on first run' },
    });
  }

  const hashed = await bcrypt.hash(DEFAULT_ADMIN_PASSWORD, 10);
  const user = await prisma.user.create({
    data: {
      username: DEFAULT_ADMIN_USERNAME,
      email: DEFAULT_ADMIN_EMAIL,
      password: hashed,
      role: 'ADMIN',
    },
    select: { id: true, username: true, email: true, role: true },
  });

  const token = jwt.sign({ sub: user.id, role: user.role }, JWT_SECRET, { expiresIn: '7d' });

  return res.json({
    success: true,
    data: { token, user, defaultPassword: true },
  });
}
```
