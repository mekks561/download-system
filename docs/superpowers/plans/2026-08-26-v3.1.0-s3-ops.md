```typescript
import bcrypt from 'bcryptjs';
import { PrismaClient } from '@prisma/client';
import { z } from 'zod';

const prisma = new PrismaClient();

function getSeedPassword(envVar: string, fallback: string): string {
  const v = process.env[envVar];
  if (!v) {
    console.warn(`[seed] ${envVar} not set, using fallback '${fallback}'`);
    return fallback;
  }
  return v;
}

export async function upsertUser(opts: {
  email: string;
  username: string;
  password: string;
  role: 'ADMIN' | 'USER';
}): Promise<void> {
  const hashed = await bcrypt.hash(opts.password, 10);
  await prisma.user.upsert({
    where: { email: opts.email },
    create: {
      email: opts.email,
      username: opts.username,
      password: hashed,
      role: opts.role,
    },
    // 关键：update 包含 password 字段（仅显式调用 seed 时覆盖）
    update: {
      password: hashed,
      updatedAt: new Date(),
    },
  });
  console.log(`[seed] upserted ${opts.email} (role=${opts.role})`);
}

async function main() {
  if (process.env.NODE_ENV === 'production') {
    console.log('[seed] production mode - skipping');
    return;
  }

  const adminPassword = getSeedPassword('SEED_ADMIN_PASSWORD', 'admin123');
  const userPassword = getSeedPassword('SEED_USER_PASSWORD', 'user123');

  await upsertUser({
    email: 'admin@dm.local',
    username: 'admin',
    password: adminPassword,
    role: 'ADMIN',
  });

  await upsertUser({
    email: 'user@dm.local',
    username: 'user',
    password: userPassword,
    role: 'USER',
  });

  console.log('[seed] done');
}

main()
  .catch((e) => {
    console.error(e);
    process.exit(1);
  })
  .finally(async () => {
    await prisma.$disconnect();
  });
```

```typescript
import { execSync } from 'node:child_process';
import * as fs from 'node:fs';

function env(key: string, fallback?: string): string {
  const v = process.env[key] ?? fallback;
  if (!v) throw new Error(`Missing env: ${key}`);
  return v;
}

function readEnvFile() {
  // Load .env manually (avoid dotenv dependency for scripts)
  const envPath = new URL('../.env', import.meta.url);
  if (fs.existsSync(envPath)) {
    const content = fs.readFileSync(envPath, 'utf-8');
    for (const line of content.split('\n')) {
      const m = line.match(/^([A-Z_]+)=(.*)$/);
      if (m && !process.env[m[1]]) process.env[m[1]] = m[2].replace(/^["']|["']$/g, '');
    }
  }
}

async function main() {
  readEnvFile();

  const dbUrl = env('DATABASE_URL');
  // Parse mysql://user:pass@host:port/db
  const m = dbUrl.match(/^mysql:\/\/([^:]+):([^@]*)@([^:]+):(\d+)\/(.+)$/);
  if (!m) throw new Error(`Invalid DATABASE_URL: ${dbUrl}`);
  const [, user, pass, host, port, db] = m;

  const ts = new Date().toISOString().replace(/[:.]/g, '-').slice(0, -5);
  const out = process.env.OUT || `backup-${ts}.sql.gz`;

  console.log(`Exporting database "${db}" from ${host}:${port} to ${out}`);
  execSync(
    `mysqldump -u ${user} -p${pass} -h ${host} -P ${port} ${db} | gzip > ${out}`,
    { stdio: 'inherit' }
  );

  const stat = fs.statSync(out);
  console.log(`✓ Exported ${out} (${(stat.size / 1024).toFixed(1)} KB)`);
}

main().catch((e) => {
  console.error(e);
  process.exit(1);
});
```

```typescript
import { execSync } from 'node:child_process';
import * as fs from 'node:fs';

function env(key: string): string {
  const v = process.env[key];
  if (!v) throw new Error(`Missing env: ${key}`);
  return v;
}

function readEnvFile() {
  const envPath = new URL('../.env', import.meta.url);
  if (fs.existsSync(envPath)) {
    const content = fs.readFileSync(envPath, 'utf-8');
    for (const line of content.split('\n')) {
      const m = line.match(/^([A-Z_]+)=(.*)$/);
      if (m && !process.env[m[1]]) process.env[m[1]] = m[2].replace(/^["']|["']$/g, '');
    }
  }
}

async function main() {
  readEnvFile();

  const file = process.argv[2];
  if (!file) {
    console.error('Usage: db:import <backup.sql.gz>');
    process.exit(1);
  }
  if (!fs.existsSync(file)) {
    console.error(`File not found: ${file}`);
    process.exit(1);
  }

  const dbUrl = env('DATABASE_URL');
  const m = dbUrl.match(/^mysql:\/\/([^:]+):([^@]*)@([^:]+):(\d+)\/(.+)$/);
  if (!m) throw new Error(`Invalid DATABASE_URL: ${dbUrl}`);
  const [, user, pass, host, port, db] = m;

  console.log(`Importing ${file} into database "${db}" at ${host}:${port}`);
  console.log('⚠️  This will OVERWRITE existing data.');
  execSync(
    `gunzip < ${file} | mysql -u ${user} -p${pass} -h ${host} -P ${port} ${db}`,
    { stdio: 'inherit' }
  );

  console.log(`✓ Imported ${file}`);
}

main().catch((e) => {
  console.error(e);
  process.exit(1);
});
```

```typescript
  GmLoginSchema,

  GmUserQuerySchema,
  GmUserUpdateSchema,
} from '@dm/shared';

export const gmRouter = Router();

// POST /gm/auth/login
gmRouter.post('/auth/login', validateBody(GmLoginSchema), async (req, res) => {
  const { username, password } = req.body;
  const gm = await prisma.gmUser.findUnique({ where: { username } });
  if (!gm || !(await bcrypt.compare(password, gm.password))) {
    return res.status(401).json({ success: false, error: { code: 'INVALID_CREDENTIALS' } });
  }
  const token = signGmToken({ id: gm.id, username: gm.username });
  res.json({ success: true, data: { token, user: { id: gm.id, username: gm.username } } });
});

// GET /gm/users
gmRouter.get('/users', requireGmAuth, validateQuery(GmUserQuerySchema), async (req, res) => {
  const { search, role, page, pageSize } = req.query as any;
  const where = {
    ...(search ? { OR: [
      { email: { contains: search } },
      { username: { contains: search } },
    ] } : {}),
    ...(role ? { role } : {}),
  };
  const [total, users] = await Promise.all([
    prisma.user.count({ where }),
    prisma.user.findMany({
      where,
      select: { id: true, email: true, username: true, role: true, isActive: true, createdAt: true },
      skip: (page - 1) * pageSize,
      take: pageSize,
      orderBy: { createdAt: 'desc' },
    }),
  ]);
  res.json({ success: true, data: { users, total, page, pageSize } });
});

// POST /gm/users/:id/reset-password
gmRouter.post('/users/:id/reset-password', requireGmAuth, validateParams(z.object({ id: z.coerce.number() })), validateBody(GmResetPasswordSchema), async (req, res) => {
  const { id } = req.params;
  const { newPassword } = req.body;
  const hashed = await bcrypt.hash(newPassword, 10);
  await prisma.user.update({ where: { id }, data: { password: hashed } });
  res.json({ success: true, data: null });
});

// PATCH /gm/users/:id
gmRouter.patch('/users/:id', requireGmAuth, validateParams(z.object({ id: z.coerce.number() })), validateBody(GmUserUpdateSchema), async (req, res) => {
  const { id } = req.params;
  await prisma.user.update({ where: { id }, data: req.body });
  res.json({ success: true, data: null });
```

```typescript
import { Request, Response, NextFunction } from 'express';
import jwt from 'jsonwebtoken';

const GM_JWT_SECRET = process.env.GM_JWT_SECRET || 'dev-gm-secret';

export interface GmTokenPayload {
  id: number;
  username: string;
}

export function signGmToken(payload: GmTokenPayload): string {
  return jwt.sign(payload, GM_JWT_SECRET, { expiresIn: '8h' });
}

export function requireGmAuth(req: Request, res: Response, next: NextFunction) {
  const header = req.headers.authorization;
  if (!header?.startsWith('Bearer ')) {
    return res.status(401).json({ success: false, error: { code: 'UNAUTHORIZED' } });
  }
  const token = header.slice(7);
  try {
    const payload = jwt.verify(token, GM_JWT_SECRET) as GmTokenPayload;
    (req as any).gmUser = payload;
    next();
  } catch {
    return res.status(401).json({ success: false, error: { code: 'INVALID_TOKEN' } });
  }
}
```

```typescript
const API_BASE = '/api/gm';

function getToken(): string | null {
  return localStorage.getItem('gm_token');
}

async function request<T>(path: string, opts: RequestInit = {}): Promise<T> {
  const token = getToken();
  const res = await fetch(`${API_BASE}${path}`, {
    ...opts,
    headers: {
      'Content-Type': 'application/json',
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...opts.headers,
    },
  });
  const json = await res.json();
  if (!json.success) throw new Error(json.error?.message || 'API error');
  return json.data as T;
}

export const api = {
  login: (username: string, password: string) =>
    request<{ token: string; user: { id: number; username: string } }>('/auth/login', {
      method: 'POST',
      body: JSON.stringify({ username, password }),
    }),

  dashboard: () =>
    request<{ userCount: number; downloadCount: number; uploadCount: number; fileCount: number; recentDownloads: any[] }>('/dashboard'),

  listUsers: (params: { page: number; pageSize: number; search?: string }) => {
    const q = new URLSearchParams({ page: String(params.page), pageSize: String(params.pageSize) });
    if (params.search) q.set('search', params.search);
    return request<{ users: any[]; total: number; page: number; pageSize: number }>(`/users?${q}`);
  },

  resetPassword: (userId: number, newPassword: string) =>
    request<null>(`/users/${userId}/reset-password`, {
      method: 'POST',
      body: JSON.stringify({ newPassword }),
    }),

  listDownloads: (params: { page: number; pageSize: number; userId?: number }) => {
    const q = new URLSearchParams({ page: String(params.page), pageSize: String(params.pageSize) });
    if (params.userId) q.set('userId', String(params.userId));
    return request<{ downloads: any[]; total: number; page: number; pageSize: number }>(`/downloads?${q}`);
  },

  cancelDownload: (id: number) =>
    request<null>(`/downloads/${id}/cancel`, { method: 'POST' }),
};
```

```tsx
        </thead>
        <tbody>
          {downloads.map((d) => (
            <tr key={d.id}>
              <td style={{ padding: 8 }}>{d.id}</td>
              <td style={{ padding: 8 }}>{d.user?.username}</td>
              <td style={{ padding: 8, maxWidth: 200, overflow: 'hidden', textOverflow: 'ellipsis' }}>{d.url}</td>
              <td style={{ padding: 8 }}>{d.status}</td>
              <td style={{ padding: 8 }}>{new Date(d.createdAt).toLocaleString()}</td>
              <td style={{ padding: 8 }}>
                {d.status !== 'CANCELLED' && d.status !== 'COMPLETED' && (
                  <button onClick={() => cancel(d.id)}>Cancel</button>
                )}
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}
```
