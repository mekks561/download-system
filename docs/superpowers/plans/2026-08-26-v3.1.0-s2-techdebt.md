```typescript
/**
 * In-process memory cache with TTL.
 * Replaces Redis-backed cache from v3.0.0 (YAGNI: Redis was never enabled in production).
 * Public API matches v3.0.0 CacheService — call sites unchanged.
 */
export class MemoryCache<K = string, V = unknown> {
  private store = new Map<K, { v: V; exp: number }>();

  get(k: K): V | undefined {
    const e = this.store.get(k);
    if (!e) return undefined;
    if (e.exp < Date.now()) {
      this.store.delete(k);
      return undefined;
    }
    return e.v;
  }

  set(k: K, v: V, ttlMs: number): void {
    this.store.set(k, { v, exp: Date.now() + ttlMs });
  }

  has(k: K): boolean {
    const e = this.store.get(k);
    if (!e) return false;
    if (e.exp < Date.now()) {
      this.store.delete(k);
      return false;
    }
    return true;
  }

  del(k: K): void {
    this.store.delete(k);
  }

  clear(): void {
    this.store.clear();
  }
}

// Backward-compatible default export (v3.0.0 used a singleton CacheService)
export const cache = new MemoryCache<string, unknown>();
```
