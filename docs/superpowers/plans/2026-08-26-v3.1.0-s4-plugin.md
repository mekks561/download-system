import type { Permission } from './permissions.js';

export interface PluginManifest {
  id: string;
  name: string;
  version: string;
  author: string;
  description: string;
  type: 'download-adapter' | 'script' | 'ui';
  permissions: Permission[];
  entry: string;
}

// Download protocol adapter
export interface DownloadAdapter {
  canHandle(url: URL): boolean;
  createTask(url: URL, opts: DownloadOpts): Promise<DownloadTask>;
  onProgress(taskId: string, cb: (p: Progress) => void): Unsubscribe;
  pause(taskId: string): Promise<void>;
  resume(taskId: string): Promise<void>;
  cancel(taskId: string): Promise<void>;
}

export interface DownloadOpts {
  downloadDir: string;
  maxConcurrent: number;
  speedLimitKBps: number;
  headers?: Record<string, string>;
}

export interface DownloadTask {
  id: string;
  url: string;
  state: 'queued' | 'downloading' | 'paused' | 'completed' | 'cancelled' | 'error';
  bytesDownloaded: number;
  bytesTotal: number;
}

export interface Progress {
  taskId: string;
  bytesDownloaded: number;
  bytesTotal: number;
  speedKBps: number;
}

export type Unsubscribe = () => void;

// Script plugin
export interface DownloadContext {
  taskId: string;
  url: string;
  filename: string;
  userId: number;
}

export interface FileContext {
  fileId: string;
  path: string;
  size: number;
  mimeType: string;
}

export interface ScheduleContext {
  trigger: 'cron' | 'interval' | 'manual';
  timestamp: number;
}

export interface ScriptPlugin {
  hooks?: Partial<{
    onDownloadStart(ctx: DownloadContext): Promise<void> | void;
    onDownloadComplete(ctx: DownloadContext): Promise<void> | void;
    onDownloadError(ctx: DownloadContext, err: Error): Promise<void> | void;
    onFileSaved(ctx: FileContext): Promise<void> | void;
    onSchedule(ctx: ScheduleContext): Promise<void> | void;
  }>;
}

// UI plugin
export interface MenuContribution {
  id: string;
  label: string;
  icon?: string;
  onClick: () => void;
  order?: number;
}

export interface PageContribution {
  id: string;
  path: string;
  label: string;
  component: React.ComponentType;
  order?: number;
}

export interface WidgetContribution {
  id: string;
  label: string;
  component: React.ComponentType;
  location: 'dashboard' | 'sidebar';
  order?: number;
}

export interface UIPlugin {
  readonly contributions?: {
    menus?: MenuContribution[];
    pages?: PageContribution[];
    widgets?: WidgetContribution[];
  };
}

// Plugin lifecycle (all plugins implement)
export interface PluginLifecycle<TCtx = PluginContext> {
  activate(ctx: TCtx): Promise<void> | void;
  deactivate(ctx: TCtx): Promise<void> | void;
}

// PluginContext: passed to activate(ctx)
export interface HostAPI {
  logger: { info(msg: string): void; warn(msg: string): void; error(msg: string): void };
  showNotification(message: string): void;
  emit(event: string, payload?: unknown): void;
  on(event: string, handler: (payload: unknown) => void): Unsubscribe;
}

export interface PluginContext<TType extends PluginManifest['type'] = 'ui'> {
  host: HostAPI;
  registerMenu: (m: MenuContribution) => void;
  registerPage: (p: PageContribution) => void;
  registerWidget: (w: WidgetContribution) => void;
  manifest: PluginManifest;
}

// Composite type: what plugin authors export as default
export type Plugin<TType extends PluginManifest['type'] = 'ui'> =
  PluginLifecycle<PluginContext<TType>> &
  (TType extends 'download-adapter' ? DownloadAdapter :
   TType extends 'script' ? ScriptPlugin :
   TType extends 'ui' ? UIPlugin : never);
```

```typescript
export type Permission =
  | { type: 'network'; hosts: string[] }
  | { type: 'fs'; paths: string[]; mode: 'r' | 'w' | 'rw' }
  | { type: 'env'; keys: string[] }
  | { type: 'spawn'; commands: string[] }
  | { type: 'ui'; inject: ('menu' | 'page' | 'widget')[] }
  | { type: 'api'; endpoints: string[] };

export function hasNetworkPermission(perms: Permission[], host: string): boolean {
  const netPerms = perms.filter((p): p is Extract<Permission, { type: 'network' }> => p.type === 'network');
  return netPerms.some((p) => p.hosts.includes('*') || p.hosts.includes(host));
}

export function hasFsPermission(perms: Permission[], path: string, mode: 'r' | 'w'): boolean {
  const fsPerms = perms.filter((p): p is Extract<Permission, { type: 'fs' }> => p.type === 'fs');
  return fsPerms.some((p) => {
    const pathAllowed = p.paths.some((allowed) => path.startsWith(allowed));
    if (!pathAllowed) return false;
    if (mode === 'r') return p.mode === 'r' || p.mode === 'rw';
    if (mode === 'w') return p.mode === 'w' || p.mode === 'rw';
    return false;
  });
}

export function hasUiPermission(perms: Permission[], scope: 'menu' | 'page' | 'widget'): boolean {
  const uiPerms = perms.filter((p): p is Extract<Permission, { type: 'ui' }> => p.type === 'ui');
  return uiPerms.some((p) => p.inject.includes(scope));
}

export function hasSpawnPermission(perms: Permission[], command: string): boolean {
  const spawnPerms = perms.filter((p): p is Extract<Permission, { type: 'spawn' }> => p.type === 'spawn');
  return spawnPerms.some((p) => p.commands.includes(command));
}

export function isPermissionGranted(granted: Permission[], requested: Permission): boolean {
  return granted.some((p) => JSON.stringify(p) === JSON.stringify(requested));
}
```

# 01 - Plugin Architecture

## Overview

Download Manager v3.1.0 introduces a plugin system allowing third-party extensions. Plugins come in three flavors:

## Plugin Types

### 1. Download Adapter (`type: 'download-adapter'`)
Implements the `DownloadAdapter` interface to add a new download protocol (e.g., magnet, FTP, ytdlp). Runs in a worker thread sandbox. Each task is a long-lived object managed by the adapter.

### 2. Script Plugin (`type: 'script'`)
Listens for host lifecycle events (`onDownloadStart`, `onDownloadComplete`, `onDownloadError`, `onFileSaved`, `onSchedule`). Stateless hooks invoked by host on event broadcast.

### 3. UI Plugin (`type: 'ui'`)
Injects menus, pages, or widgets into the main UI. Runs in the main React process (subject to CSP). Uses `ctx.registerMenu/registerPage/registerWidget`.

## Package Boundaries

- `packages/plugin-sdk`: Type-only package, no runtime. Published to npm so plugin authors can install as devDependency for type checking.
- `apps/api/src/plugins/`: Runtime implementation on host side. Contains `PluginHost` (lifecycle) and `SandboxRunner` (worker_threads).
- `plugins/`: Example plugins. v3.1.0 ships `hello-world` as reference.

## Host Architecture

```
┌──────────────────────────────────────────────┐
│ Electron main process                       │
│  ├─ apps/desktop (Electron 32)              │
│  ├─ apps/api (Express on 5001)              │
│  │   ├─ PluginHost                          │
│  │   │   ├─ loads installed.json            │
│  │   │   ├─ forks SandboxRunner per plugin  │
│  │   │   └─ broadcasts events to workers    │
│  │   └─ apps/web (React 19 on 5173)         │
│  └─ MariaDB                                 │
└──────────────────────────────────────────────┘
```

## SDK Boundary

`@dm/plugin-sdk` exports only types and pure helpers (e.g., permission checkers). It does NOT import Express/Prisma/Electron. This allows:
- Plugin authors to install only the SDK without host dependencies
- Independent `npm publish` of SDK to npm registry
- No circular dependency between host and SDK

## Why Three Plugin Types?

Single-interface plugins would force UI to be dynamic-imported in worker (impossible: workers don't have DOM). Separating concerns:
- Worker: heavy CPU/IO (download protocols, scripts that fs/require)
- Main: UI rendering with React

See `02-sandbox.md` for worker boundaries.

# 02 - Sandbox

## Worker Threads Selection

Three sandbox options were evaluated:

| Option | Isolation | Perf | Maintenance | Complexity |
|---|---|---|---|---|
| vm2 | High | Medium | Deprecated | Low |
| isolated-vm | Very High | High (V8 isolate) | Active | High (C++ addon) |
| **worker_threads** | Medium | High | Native | Medium |

We select `worker_threads` because:
- Native Node.js API, no native deps to build
- Electron already bundles Node.js
- Performance sufficient for plugin workloads (typically IO-bound)
- Isolation adequate: separate V8 context, separate event loop, restricted globals

## Implementation: SandboxRunner

`apps/api/src/plugins/SandboxRunner.ts` wraps `worker_threads.Worker`:

```typescript
import { Worker } from 'node:worker_threads';
import path from 'node:path';

export class SandboxRunner {
  private worker: Worker;

  constructor(pluginPath: string, sandbox: PluginContext) {
    this.worker = new Worker(pluginPath, {
      workerData: { sandbox },
      eval: false,
    });
  }

  async activate(): Promise<void> {
    return new Promise((resolve, reject) => {
      this.worker.once('message', (msg) => msg === 'activated' ? resolve() : reject(new Error('Unexpected')));
      this.worker.postMessage({ type: 'activate' });
    });
  }

  async broadcast(event: string, payload: unknown): Promise<void> {
    this.worker.postMessage({ type: 'event', event, payload });
  }

  async terminate(): Promise<void> {
    await this.worker.terminate();
  }
}
```

## Plugin-side entry point

Plugins ship `index.js` with:

```js
import { parentPort, workerData } from 'node:worker_threads';

const ctx = workerData.sandbox;

parentPort.on('message', async (msg) => {
  if (msg.type === 'activate') {
    await plugin.activate(ctx);
    parentPort.postMessage('activated');
  } else if (msg.type === 'event') {
    // dispatch via host.on() handlers
    ctx.host._dispatchEvent(msg.event, msg.payload);
  } else if (msg.type === 'deactivate') {
    await plugin.deactivate(ctx);
    parentPort.postMessage('deactivated');
    process.exit(0);
  }
});
```

## Boundary: What Runs in Worker vs Main

| Concern | Worker | Main React process |
|---|---|---|
| Download protocol logic | ✓ | ✗ |
| fs / require / spawn | ✓ (scoped) | ✗ |
| Network requests | ✓ (scoped fetch) | ✓ (host's fetch) |
| DOM access | ✗ | ✓ |
| React components | ✗ | ✓ |
| Event hook execution | ✓ | ✗ (broadcast via postMessage) |

## Why Not vm2?

vm2 was deprecated in 2023 after multiple unpatchable sandbox-escape CVEs. Avoid.

## Why Not isolated-vm?

isolated-vm requires a C++ addon build step, complicates Electron packaging. The performance benefit (V8 isolate vs full V8 instance) is marginal for plugin workloads. Defer to v3.2.0+ only if profiling shows need.

## Limitations

- Workers cannot use `import` from main process modules directly (must use absolute paths or bundled deps)
- SharedArrayBuffer not used (plugins don't need shared memory)
- Cross-worker communication via postMessage (JSON-serializable only)
```

# 03 - Permission Model

## Permission Types

Six declarative permission types defined in `packages/plugin-sdk/src/permissions.ts`:

| Type | Example | Purpose |
|---|---|---|
| `network` | `{ hosts: ['api.github.com'] }` | Whitelist domains for fetch |
| `fs` | `{ paths: ['/tmp/dl'], mode: 'rw' }` | Whitelist file system paths |
| `env` | `{ keys: ['API_KEY'] }` | Whitelist env vars accessible |
| `spawn` | `{ commands: ['ytdlp'] }` | Whitelist sub-processes |
| `ui` | `{ inject: ['menu', 'page'] }` | UI injection scopes |
| `api` | `{ endpoints: ['/api/downloads'] }` | Whitelist host API endpoints |

## Declarative in manifest

```json
{
  "permissions": [
    { "type": "network", "hosts": ["api.github.com"] },
    { "type": "fs", "paths": ["${DOWNLOAD_DIR}"], "mode": "rw" },
    { "type": "spawn", "commands": ["ytdlp", "ffmpeg"] }
  ]
}
```

Manifest declares all required permissions at install time. User sees this list and must explicitly approve.

## Runtime enforcement

`PluginHost` builds scoped APIs from manifest.permissions:

```typescript
const scopedFetch = makeScopedFetch(manifest.permissions);  // 403 for non-whitelisted host
const scopedFs = makeScopedFs(manifest.permissions);        // EACCES for non-whitelisted path
const scopedRequire = makeScopedRequire(manifest.permissions);
const scopedSpawn = makeScopedSpawn(manifest.permissions);  // EPERM for non-whitelisted command
```

These are injected into the worker via `workerData.sandbox`.

## User Authorization Flow

1. User clicks "Install" in plugin marketplace UI
2. Host downloads plugin bundle (zip with manifest.json + entry.js + assets)
3. Host parses manifest.json
4. UI displays the permissions list with human-readable descriptions
5. User clicks "Approve"
6. Host writes entry to `plugins/installed.json`:
   ```json
   { "hello-world": { "version": "1.0.0", "path": "plugins/hello-world/", "approvedAt": "..." } }
   ```
7. On next host startup, PluginHost loads the entry

## Denial Flow

If user clicks "Deny":
- Bundle is deleted from disk
- No entry written to `installed.json`
- Plugin never loads

## Runtime Permission Check

Helpers in `packages/plugin-sdk`:

- `hasNetworkPermission(perms, host)`
- `hasFsPermission(perms, path, mode)`
- `hasUiPermission(perms, scope)`
- `hasSpawnPermission(perms, command)`

Used by `apps/api/src/plugins/scoped-*.ts` to enforce at call site.

## Future: Runtime Permission Upgrade

v3.2.0 may add runtime permission upgrade (e.g., a UI plugin requests `fs` access dynamically). For v3.1.0, all permissions must be declared at install time.

## Security Considerations

- Permissions are additive: declaring more permissions never reduces isolation
- Wildcards (`hosts: ['*']`) allow all hosts but still go through scoped fetch (so logging/auditing works)
- `fs` mode `'r'` is read-only; plugins cannot write without `'w'` or `'rw'`
- `spawn` only allows exact command name (no shell expansion)
```

# 04 - Plugin Lifecycle

## States

A plugin transitions through these states:

```
[installed] → [loaded] → [activated] → [deactivated] → [uninstalled]
                                                      ↘ [error]
```

## Install Flow

1. User approves manifest (see 03-permissions.md)
2. Host unzips bundle to `plugins/<id>/`
3. Validates signature (v3.2.0)
4. Writes entry to `plugins/installed.json`
5. State: `installed`

## Host Startup

For each entry in `installed.json`:

1. Read `manifest.json`
2. Resolve entry file path
3. Fork `SandboxRunner` (worker thread)
4. Send `activate` message with sandboxed `ctx`
5. Plugin's `activate(ctx)` runs:
   - UI plugins call `ctx.registerMenu(...)` etc.
   - Script plugins attach `ctx.host.on('event', handler)`
   - Download adapters register with `host.downloadRegistry.register(this)`
6. State: `activated`

## Event Broadcast

Host emits events on lifecycle changes:

- `download:start` (with DownloadContext)
- `download:complete` (with DownloadContext)
- `download:error` (with DownloadContext + Error)
- `file:saved` (with FileContext)
- `schedule:trigger` (with ScheduleContext)

`PluginHost.broadcast(event, payload)` posts to all workers via `SandboxRunner.broadcast`. Workers invoke registered handlers.

## Deactivate Flow

User clicks "Disable" in marketplace UI, or host shuts down:

1. Host sends `deactivate` message to worker
2. Plugin's `deactivate(ctx)` runs (cleanup: close files, remove timers)
3. Worker exits
4. State: `deactivated` (still installed, can be re-enabled)

## Uninstall Flow

User clicks "Uninstall":

1. If `activated`, run deactivate first
2. Delete `plugins/<id>/` directory
3. Remove entry from `installed.json`
4. State: `uninstalled`

## Error Recovery

If worker crashes (uncaught exception):

1. `Worker.on('error')` fires in `SandboxRunner`
2. `PluginHost` logs the error
3. Marks plugin as `error` state in `installed.json`
4. Does NOT auto-restart (user intervention required)
5. UI shows warning: "Plugin X crashed, click to restart"

## Persistence

`installed.json` schema:

```json
{
  "hello-world": {
    "version": "1.0.0",
    "path": "plugins/hello-world/",
    "enabled": true,
    "approvedAt": "2026-08-26T12:00:00.000Z",
    "state": "activated",
    "lastError": null
  }
}
```

## Concurrency

- Multiple plugins run in parallel (each in own worker)
- Events broadcast to all workers concurrently
- Host does not serialize event dispatch
- Workers cannot directly message each other (must go through host via `ctx.host.emit`)
```

```typescript
import { Worker } from 'node:worker_threads';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import type { PluginContext } from '@dm/plugin-sdk';

const __dirname = path.dirname(fileURLToPath(import.meta.url));

export interface SandboxRunnerOptions {
  pluginPath: string;
  context: PluginContext;
}

export class SandboxRunner {
  private worker: Worker;
  private eventHandlers = new Map<string, ((payload: unknown) => void)[]>();

  constructor(pluginPath: string, context: PluginContext) {
    this.worker = new Worker(pluginPath, {
      workerData: { sandbox: context },
      eval: false,
    });

    this.worker.on('message', (msg: any) => {
      if (msg.type === 'event' && msg.event) {
        const handlers = this.eventHandlers.get(msg.event) ?? [];
        handlers.forEach((h) => h(msg.payload));
      }
    });
  }

  async activate(): Promise<boolean> {
    return new Promise((resolve, reject) => {
      const timeout = setTimeout(() => reject(new Error('activate timeout')), 5000);
      this.worker.once('message', (msg) => {
        clearTimeout(timeout);
        if (msg === 'activated') resolve(true);
        else reject(new Error(`Unexpected message: ${msg}`));
      });
      this.worker.once('error', (err) => {
        clearTimeout(timeout);
        reject(err);
      });
      this.worker.postMessage({ type: 'activate' });
    });
  }

  async broadcast(event: string, payload: unknown): Promise<void> {
    this.worker.postMessage({ type: 'event', event, payload });
  }

  async deactivate(): Promise<void> {
    return new Promise((resolve) => {
      this.worker.once('message', (msg) => {
        if (msg === 'deactivated') resolve();
      });
      this.worker.postMessage({ type: 'deactivate' });
    });
  }

  async terminate(): Promise<void> {
    await this.worker.terminate();
  }
}

```typescript
import fs from 'node:fs/promises';
import path from 'node:path';
import { SandboxRunner } from './SandboxRunner.js';
import type { PluginManifest, PluginContext, HostAPI, MenuContribution } from '@dm/plugin-sdk';

interface InstalledEntry {
  version: string;
  path: string;
  enabled: boolean;
  approvedAt: string;
  state: 'installed' | 'activated' | 'deactivated' | 'error' | 'uninstalled';
  lastError: string | null;
}

interface PluginHostOptions {
  pluginsDir: string;
  installedJsonPath: string;
}

export class PluginHost {
  private installed = new Map<string, { entry: InstalledEntry; runner?: SandboxRunner; manifest?: PluginManifest }>();
  private menus: MenuContribution[] = [];
  private options: PluginHostOptions;

  constructor(options: PluginHostOptions) {
    this.options = options;
  }

  async loadAll(): Promise<void> {
    const raw = await fs.readFile(this.options.installedJsonPath, 'utf-8').catch(() => '{}');
    const data = JSON.parse(raw) as Record<string, InstalledEntry>;

    for (const [id, entry] of Object.entries(data)) {
      if (!entry.enabled) continue;
      try {
        await this.loadPlugin(id, entry);
      } catch (err: any) {
        entry.state = 'error';
        entry.lastError = err.message;
        console.error(`[PluginHost] failed to load ${id}:`, err);
      }
    }
    await this.persistInstalled();
  }

  private async loadPlugin(id: string, entry: InstalledEntry): Promise<void> {
    const pluginPath = path.join(this.options.pluginsDir, entry.path, 'index.js');
    const manifestPath = path.join(this.options.pluginsDir, entry.path, 'manifest.json');
    const manifest = JSON.parse(await fs.readFile(manifestPath, 'utf-8')) as PluginManifest;

    const ctx: PluginContext = {
      host: this.makeHostAPI(id),
      registerMenu: (m) => {
        if (this.hasUiPermission(manifest, 'menu')) {
          this.menus.push(m);
          console.log(`[PluginHost] ${id} registered menu: ${m.label}`);
        } else {
          console.warn(`[PluginHost] ${id} denied menu registration (no ui.inject.menu permission)`);
        }
      },
      registerPage: () => {},
      registerWidget: () => {},
      manifest,
    };

    const runner = new SandboxRunner(pluginPath, ctx);
    await runner.activate();

    this.installed.set(id, { entry, runner, manifest });
    entry.state = 'activated';
    entry.lastError = null;
  }

  private hasUiPermission(manifest: PluginManifest, scope: 'menu' | 'page' | 'widget'): boolean {
    return manifest.permissions.some(
      (p) => p.type === 'ui' && p.inject.includes(scope)
    );
  }

  private makeHostAPI(pluginId: string): HostAPI {
    return {
      logger: {
        info: (msg) => console.log(`[${pluginId}] ${msg}`),
        warn: (msg) => console.warn(`[${pluginId}] ${msg}`),
        error: (msg) => console.error(`[${pluginId}] ${msg}`),
      },
      showNotification: (message) => console.log(`[${pluginId}] notification: ${message}`),
      emit: (event, payload) => this.broadcast(event, payload),
      on: () => () => {}, // TODO: event subscription
    };
  }

  async broadcast(event: string, payload: unknown): Promise<void> {
    for (const { runner } of this.installed.values()) {
      if (runner) await runner.broadcast(event, payload);
    }
  }

  listInstalled(): Array<{ id: string; version: string; state: string }> {
    return Array.from(this.installed.entries()).map(([id, { entry }]) => ({
      id,
      version: entry.version,
      state: entry.state,
    }));
  }

  listMenus(): MenuContribution[] {
    return [...this.menus];
  }

  async uninstall(id: string): Promise<void> {
    const plugin = this.installed.get(id);
    if (plugin?.runner) {
      await plugin.runner.deactivate();
      await plugin.runner.terminate();
    }
    this.installed.delete(id);

    // Remove from disk
    const pluginDir = path.join(this.options.pluginsDir, id);
    await fs.rm(pluginDir, { recursive: true, force: true });

    await this.persistInstalled();
  }

  async shutdown(): Promise<void> {
    for (const { runner } of this.installed.values()) {
      if (runner) {
        await runner.deactivate();
        await runner.terminate();
      }
    }
    this.installed.clear();
  }

  private async persistInstalled(): Promise<void> {
    const data: Record<string, InstalledEntry> = {};
    for (const [id, { entry }] of this.installed) {
      data[id] = entry;
    }
    await fs.writeFile(this.options.installedJsonPath, JSON.stringify(data, null, 2));
  }
}
```

  // GET /api/plugins/registry/:id

    const { id } = req.params;
    try {
      const manifest = JSON.parse(
        await fs.readFile(path.join(pluginsDir, id, 'manifest.json'), 'utf-8')
      );
      res.json({ success: true, data: manifest });
    } catch {
      res.status(404).json({ success: false, error: { code: 'NOT_FOUND' } });
    }
  });

  // GET /api/plugins/installed
  router.get('/installed', async (_req, res) => {
    const installedPath = path.join(pluginsDir, 'installed.json');
    const raw = await fs.readFile(installedPath, 'utf-8').catch(() => '{}');
    res.json({ success: true, data: JSON.parse(raw) });
  });

  // POST /api/plugins/install
  router.post('/install', async (req, res) => {
    const { id, version } = req.body;
    if (!id) return res.status(400).json({ success: false, error: { code: 'MISSING_ID' } });

    const manifestPath = path.join(pluginsDir, id, 'manifest.json');
    try {
      const manifest = JSON.parse(await fs.readFile(manifestPath, 'utf-8'));
      if (version && manifest.version !== version) {
        return res.status(400).json({ success: false, error: { code: 'VERSION_MISMATCH' } });
      }
      res.json({ success: true, data: { manifest } });
    } catch {
      res.status(404).json({ success: false, error: { code: 'PLUGIN_NOT_FOUND' } });
    }
  });

  // POST /api/plugins/:id/enable
  // POST /api/plugins/:id/disable
  // DELETE /api/plugins/:id
  // (skeleton - implemented in v3.2.0 with front-end UI)
  router.post('/:id/enable', async (req, res) => {
    res.status(501).json({ success: false, error: { code: 'NOT_IMPLEMENTED', message: 'Use installed.json directly in v3.1.0' } });
  });

  return router;
}
```
